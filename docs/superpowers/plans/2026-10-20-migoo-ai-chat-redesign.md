# Migoo-Style AI Chat UI Redesign Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Redesign AgriDirect's Kiko AI assistant screen to adopt the clean, modern e-commerce conversational UI inspired by TikTok Shop's Migoo assistant (iridescent glowing mascot halo, greeting bubble, suggestion pills, quick category bar, and capsule input).

**Architecture:** Refactor `KikoAiChatScreen` in `lib/mobile/screens/support/kiko_ai_chat_screen.dart` to modularize and render:
1. A minimalist header with centered title "Kiko" and subtitle "Generated by AI".
2. An iridescent sweep-gradient hero mascot halo with Kiko Carabao and soft-grey welcome bubble.
3. Role-adaptive suggestion pills and floating category shortcuts.
4. A rounded capsule bottom input composer with `+` attachment sheet, "Anything on your mind?" placeholder, and mic/send toggle.

**Tech Stack:** Flutter, Dart, GoogleFonts (Poppins & Inter), ImagePicker.

## Global Constraints
- File target: `lib/mobile/screens/support/kiko_ai_chat_screen.dart`.
- Preserve existing backend integrations: `AiService` (Gemini), `WeatherService`, and image upload for crop disease diagnosis.
- Maintain desktop web and embed mode responsiveness without layout overflow.
- Clean typography and colors: Dark text `#111827`, subtitle `#9CA3AF`, pill borders `#E5E7EB`, capsule background `#F3F4F6`, emerald accent `#059669`.

  List<String> get _migooSuggestionPrompts {
    final isFarmer = _auth.isViewingAsFarmer;
    if (isFarmer) {
      return [
        _weatherPromptLabel,
        '🌾 Gabay sa pagpepresyo ng ani',
        '🌱 Tamang planting window ng kamatis',
        '🐛 Pest and disease diagnosis guide',
        '📦 Paano gumagana ang pre-orders',
      ];
    }
    return [
      'Find best deals',
      'Snacks worth trying & fresh harvest',
      'Trending in organic produce',
      'Sun protection & farmer tools',
      'Useful kitchen storage tips',
    ];
  }

  Widget _buildSuggestionPills() {
    final prompts = _migooSuggestionPrompts;
    return Container(
      width: double.infinity,
      padding: const EdgeInsets.symmetric(horizontal: 4),
      child: Wrap(
        spacing: 8,
        runSpacing: 8,
        children: prompts.map((prompt) {
          return Material(
            color: Colors.white,
            borderRadius: BorderRadius.circular(14),
            child: InkWell(
              onTap: () => _sendMessage(prompt),
              borderRadius: BorderRadius.circular(14),
              child: Container(
                padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 10),
                decoration: BoxDecoration(
                  borderRadius: BorderRadius.circular(14),
                  border: Border.all(color: const Color(0xFFE5E7EB)),
                ),
                child: Text(
                  prompt,
                  style: GoogleFonts.inter(
                    fontSize: 13,
                    fontWeight: FontWeight.w500,
                    color: const Color(0xFF1F2937),
                  ),
                ),
              ),
            ),
          );
        }).toList(),
      ),
    );
  }

  Widget _buildCategoryShortcutBar() {
    final isFarmer = _auth.isViewingAsFarmer;
    final shortcuts = isFarmer
        ? [
            {'icon': Icons.eco_rounded, 'label': 'Farming', 'prompt': '🌾 Paano mag-list ng sariwang ani?'},
            {'icon': Icons.wb_sunny_rounded, 'label': 'Weather', 'prompt': _weatherPromptLabel},
            {'icon': Icons.receipt_long_rounded, 'label': 'Sales', 'prompt': '📊 Paano tingnan ang aking sales?'},
            {'icon': Icons.discount_rounded, 'label': 'Voucher', 'prompt': '🎟️ Paano gumawa ng shop vouchers?'},
          ]
        : [
            {'icon': Icons.shopping_bag_outlined, 'label': 'Shopping', 'prompt': '🛒 Paano mag-order ng sariwang ani?'},
            {'icon': Icons.confirmation_number_outlined, 'label': 'Voucher', 'prompt': '🎟️ Paano mag-claim ng exclusive discount vouchers?'},
            {'icon': Icons.inbox_outlined, 'label': 'Inbox', 'prompt': '📍 Paano i-track ang status ng aking order?'},
            {'icon': Icons.local_offer_outlined, 'label': 'Deals', 'prompt': 'Find best deals'},
          ];

    return Container(
      height: 40,
      margin: const EdgeInsets.only(bottom: 8),
      child: ListView.separated(
        scrollDirection: Axis.horizontal,
        padding: const EdgeInsets.symmetric(horizontal: 14),
        itemCount: shortcuts.length,
        separatorBuilder: (_, __) => const SizedBox(width: 8),
        itemBuilder: (context, idx) {
          final item = shortcuts[idx];
          return Material(
            color: Colors.white,
            borderRadius: BorderRadius.circular(16),
            child: InkWell(
              onTap: () => _sendMessage(item['prompt'] as String),
              borderRadius: BorderRadius.circular(16),
              child: Container(
                padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 7),
                decoration: BoxDecoration(
                  borderRadius: BorderRadius.circular(16),
                  border: Border.all(color: const Color(0xFFE5E7EB)),
                ),
                child: Row(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    Icon(item['icon'] as IconData, size: 16, color: const Color(0xFF374151)),
                    const SizedBox(width: 6),
                    Text(
                      item['label'] as String,
                      style: GoogleFonts.inter(
                        fontSize: 12.5,
                        fontWeight: FontWeight.w500,
                        color: const Color(0xFF1F2937),
                      ),
                    ),
                  ],
                ),
              ),
            ),
          );
        },
      ),
    );
  }
```

  PreferredSizeWidget _buildAppBar() {
    return AppBar(
      backgroundColor: Colors.white,
      elevation: 0,
      scrolledUnderElevation: 0,
      leading: IconButton(
        icon: const Icon(
          Icons.person_outline_rounded,
          color: Color(0xFF1F2937),
          size: 24,
        ),
        onPressed: () {
          if (Navigator.canPop(context)) {
            Navigator.pop(context);
          }
        },
      ),
      centerTitle: true,
      title: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          Text(
            'Kiko',
            style: GoogleFonts.poppins(
              fontWeight: FontWeight.w700,
              fontSize: 17,
              color: const Color(0xFF111827),
            ),
          ),
          Text(
            'Generated by AI',
            style: GoogleFonts.inter(
              fontSize: 11,
              fontWeight: FontWeight.w400,
              color: const Color(0xFF9CA3AF),
            ),
          ),
        ],
      ),
      actions: [
        IconButton(
          icon: const Icon(
            Icons.description_outlined,
            color: Color(0xFF1F2937),
            size: 22,
          ),
          tooltip: 'Reset or view topics',
          onPressed: _showChatOptionsSheet,
        ),
      ],
    );
  }

  Widget _buildInputComposer() {
    final hasText = _messageController.text.trim().isNotEmpty || _selectedImageBytes != null;

    return Container(
      padding: const EdgeInsets.fromLTRB(14, 8, 14, 12),
      color: Colors.white,
      child: Container(
        padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 4),
        decoration: BoxDecoration(
          color: const Color(0xFFF3F4F6),
          borderRadius: BorderRadius.circular(28),
        ),
        child: Row(
          children: [
            // + Attachment Action (Camera / Gallery for Crop Diagnosis)
            IconButton(
              icon: const Icon(Icons.add_rounded, color: Color(0xFF1F2937), size: 24),
              onPressed: _showAttachmentSheet,
              padding: EdgeInsets.zero,
              constraints: const BoxConstraints(minWidth: 36, minHeight: 36),
            ),
            const SizedBox(width: 4),

            // Text Input Field
            Expanded(
              child: TextField(
                controller: _messageController,
                textInputAction: TextInputAction.send,
                onChanged: (_) => setState(() {}),
                onSubmitted: (_) => _sendMessage(),
                style: GoogleFonts.inter(
                  fontSize: 14,
                  color: const Color(0xFF111827),
                ),
                decoration: InputDecoration(
                  hintText: 'Anything on your mind?',
                  hintStyle: GoogleFonts.inter(
                    fontSize: 14,
                    color: const Color(0xFF9CA3AF),
                  ),
                  border: InputBorder.none,
                  isDense: true,
                  contentPadding: const EdgeInsets.symmetric(vertical: 10),
                ),
              ),
            ),

            // Trailing Mic or Send Action
            if (hasText)
              Container(
                width: 34,
                height: 34,
                decoration: const BoxDecoration(
                  color: Color(0xFF059669),
                  shape: BoxShape.circle,
                ),
                child: IconButton(
                  icon: const Icon(Icons.arrow_upward_rounded, color: Colors.white, size: 19),
                  padding: EdgeInsets.zero,
                  onPressed: () => _sendMessage(),
                ),
              )
            else
              IconButton(
                icon: const Icon(Icons.mic_none_rounded, color: Color(0xFF1F2937), size: 22),
                padding: EdgeInsets.zero,
                constraints: const BoxConstraints(minWidth: 36, minHeight: 36),
                onPressed: () {
                  ScaffoldMessenger.of(context).showSnackBar(
                    const SnackBar(
                      content: Text('Voice input is coming soon! You can type in the box.'),
                      duration: Duration(seconds: 2),
                    ),
                  );
                },
              ),
          ],
        ),
      ),
    );
  }
```
